main                : body

body                : (NEWLINE* expr NEWLINE+)* END

nodes               : expr
                    : LBRACE body RBRACE

controls            : KW:if expr NEWLINE* LBRACE nodes RBRACE
                        (NEWLINE* KW:elseif expr NEWLINE* LBRACE nodes RBRACE)*
                        (NEWLINE* KW:else nodes)?
                    : KW:while expr NEWLINE* LBRACE nodes RBRACE
                    : KW:for term = expr KW:to expr (KW:step expr)? nodes

expr                : keyword

keyword             : KW:wln expr
                    : KW:w expr
                    : KW:rei expr
                    : KW:rech expr
                    : KW:rek expr
                    : KW:if expr KW:then expr KW:else expr
                    : KW:cpy term COMMA expr
                    : KW:break
                    : Kw:next
                    : comp

comp                : KW:lt expr, expr                  # check value less than other
                    : KW:gt expr, expr                  # check value greater than other
                    : KW:eq expr, expr                  # equality check
                    : KW:and expr, expr                 # logical and
                    : KW:or expr, expr                  # logical or
                    : KW:xor expr, expr                 # logical or (exclusive)
                    : term                              # term

term                : DOT factor                        # register address
                    : factor                            # factor

factor              : INT|STRING                        # values
                    : LPAREN expr RPAREN                # sub-expressions
                    : PLUS|MINUS factor                 # unary operation
                    : MUL factor                        # dereference register
                    : INCREMENT|DECREMENT factor        # increment or decrement
                    : KW:not factor                     # logical not
